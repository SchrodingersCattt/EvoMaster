<!-- MessageBus emit/emit_nowait signatures from matmaster/core/bus.py -->
```python
class MessageBus:
    async def emit(self, event: BusEvent) -> None:
        """Emit event (non-blocking for unbounded queue). Must be called from within the event loop."""
        self._queue.put_nowait(event)

    def emit_nowait(self, event: BusEvent) -> None:
        """Thread-safe sync emit for cross-thread callers (service layer)."""
        if self._loop is not None and self._loop.is_running():
            self._loop.call_soon_threadsafe(self._queue.put_nowait, event)
        else:
            self._queue.put_nowait(event)
```
