```python
class Session(Protocol):
    @property
    def is_open(self) -> bool: ...
    def open(self) -> None: ...
    def close(self) -> None: ...
    def exec_bash(self, command: str, timeout: int | None = None, is_input: bool = False, stop_event: threading.Event | Any | None = None) -> dict[str, Any]: ...
    def read_file(self, path: str, encoding: str = "utf-8") -> str: ...
    def write_file(self, path: str, content: str, encoding: str = "utf-8") -> None: ...

```
