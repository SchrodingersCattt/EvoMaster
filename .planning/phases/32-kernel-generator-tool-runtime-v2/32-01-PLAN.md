    ```python
    from __future__ import annotations
    import enum
    from typing import Literal
    from pydantic import BaseModel, ConfigDict

    class ToolPlane(str, enum.Enum):
        SESSION_SHELL = "session_shell"
        SESSION_FS = "session_fs"
        CONTROL_PLANE = "control_plane"
        EXTERNAL_SERVICE = "external_service"

    class SessionCapabilities(BaseModel):
        model_config = ConfigDict(frozen=True)
        shell_persistence: Literal["stateless", "persistent"] = "stateless"
        shell_input: bool = False
        file_ops: Literal["native", "sftp"] = "native"
        upload_support: bool = False
        exec_cancel: bool = False

    class RuntimeTopology(BaseModel):
        model_config = ConfigDict(frozen=True)
        session_kind: str  # "local" | "ssh" | "docker"
        control_root: str  # 控制平面工作目录
        workspace_root: str  # session 工作区根路径
        active_planes: frozenset[ToolPlane] = frozenset()
        session_capabilities: SessionCapabilities | None = None
    ```

    **2. 创建 matmaster/types/tool_spec.py** (per TOBJ-04, TOBJ-05, TOBJ-06, TOBJ-07):

    ```python
    from __future__ import annotations
    from collections.abc import Awaitable, Callable
    from dataclasses import dataclass
    from typing import Any, Literal
    from pydantic import BaseModel, ConfigDict, Field
    from matmaster.types.topology import ToolPlane

    class ToolSpec(BaseModel):
        model_config = ConfigDict(frozen=True)
        tool_name: str
        description: str = ""
        args_schema: dict[str, Any] = Field(default_factory=dict)
        source: str = "unknown"
        capabilities: frozenset[str] = frozenset()
        effect_level: str = "local_mutation"  # "pure_read" | "local_mutation" | "external_write"
        exposed_to_model: bool = True
        fast_path_eligible: bool = False

    class ResourceClaim(BaseModel):
        model_config = ConfigDict(frozen=True)
        resource_id: str
        mode: Literal["exclusive", "shared_read", "counted"]
        limit: int | None = None  # 仅 mode="counted" 时有意义

    class ToolBinding(BaseModel):
        model_config = ConfigDict(frozen=True)
        binding_key: str  # "{plane}:{tool_name}"
        plane: ToolPlane
        resource_claims: tuple[ResourceClaim, ...] = ()
        state_mode: str = "stateless"  # "stateless" | "turn_scoped" | "session_scoped"
        stop_mode: str = "immediate"  # "immediate" | "graceful" | "detached"

    from matmaster.tools.tool_result import ToolResult

    @dataclass(frozen=True)
    class ToolInstance:
        tool_spec: ToolSpec
        tool_binding: ToolBinding
        tool_executor: Callable[[dict[str, Any]], Awaitable[ToolResult]]
    ```
