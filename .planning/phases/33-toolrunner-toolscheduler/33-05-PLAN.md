            Args:

                tool_name=tool.name,

                source=source,
                effect_level=effect_level,
                fast_path_eligible=fast_path,
            )

                binding_key=f"{plane.value}:{tool.name}",
                plane=plane,
                resource_claims=claims,
            )
            return ToolInstance(
                tool_spec=spec,
                tool_binding=binding,
                tool_executor=tool.execute,
            )

    完整重写 tool_catalog.py：

    ```python
    """ToolCatalog -- facade over ToolRegistry with ToolCompiler integration.

    Provides base+overlay structure with version tracking for lazy MCP tool
    injection. get_tool() returns pre-compiled ToolInstance objects.

    Per advancement plan D-05: ToolCompiler is called at registration time,
    ToolCatalog stores compiled ToolInstance objects, FullToolRunner consumes
    them directly without runtime resolution.
    """

    from __future__ import annotations

    from typing import TYPE_CHECKING, Any

    from matmaster.tools.tool_registry import Tool, ToolRegistry
    from matmaster.types.tool_spec import ToolInstance

    if TYPE_CHECKING:
        from matmaster.tools.tool_compiler import ToolCompiler
        from matmaster.types.topology import RuntimeTopology


    class ToolCatalog:
        """Facade over ToolRegistry with ToolCompiler-based instance creation.

        version starts at 0 and increments on each register_overlay() call.
        Kernel compares version each turn to decide whether to refresh
        tool_definitions for the LLM.
        """

        def __init__(
            self,
            registry: ToolRegistry,
            compiler: ToolCompiler | None = None,
            topology: RuntimeTopology | None = None,
        ) -> None:
            self._registry = registry
            self._compiler = compiler
            self._topology = topology
            self._version: int = 0
            self._compiled: dict[str, ToolInstance] = {}

        @property
        def version(self) -> int:
            """Current catalog version. Incremented on overlay registrations."""
            return self._version

        @property
        def registry(self) -> ToolRegistry:
            """Expose internal registry for backward-compatible access paths."""
            return self._registry

        def register_overlay(self, tool: Tool, *, source: str = "mcp") -> None:
            """Register a tool in the overlay layer (MCP/skill lazy injection).

            If ToolCompiler is available, compiles the tool into ToolInstance
            at registration time. Increments version so Kernel can detect
            tool set changes.
            """
            self._registry.register(tool, source=source)
            if self._compiler is not None and self._topology is not None:
                instance = self._compiler.compile(
                    tool, self._topology, source=source
                )
                self._compiled[tool.name] = instance
            self._version += 1

        def get_tool(self, tool_name: str) -> ToolInstance | None:
            """Look up tool and return compiled ToolInstance.

            If ToolCompiler is available, uses compiler to produce ToolInstance
            (caching result). Falls back to inline construction if no compiler
            is configured (backward compatibility with Phase 32 tests).
            """
            # Check cache first
            if tool_name in self._compiled:
                return self._compiled[tool_name]

            raw_tool = self._registry._tools.get(tool_name)
            if raw_tool is None:
                return None

            if self._compiler is not None and self._topology is not None:
                source = self._registry._sources.get(tool_name, "unknown")
                instance = self._compiler.compile(
                    raw_tool, self._topology, source=source
                )
                self._compiled[tool_name] = instance
                return instance

            # Fallback: no compiler configured (backward compat)
            return self._compile_fallback(raw_tool, tool_name)

        def _compile_fallback(
            self, raw_tool: Tool, tool_name: str
        ) -> ToolInstance:
            """Inline compilation fallback when no ToolCompiler is configured.

            This preserves backward compatibility with code that constructs
            ToolCatalog(registry) without a compiler (e.g., Phase 32 tests).
            Uses the same BUILTIN_RULES data via import.
            """
            from matmaster.tools.tool_compiler import ToolCompiler as _TC
            from matmaster.types.topology import RuntimeTopology as _RT

            # Construct minimal topology for fallback compilation
            fallback_topology = _RT(
                session_kind="local",
                control_root=".",
                workspace_root=".",
            )
            compiler = _TC()
            source = self._registry._sources.get(tool_name, "unknown")
            return compiler.compile(raw_tool, fallback_topology, source=source)

        def build_definitions(self) -> list[dict[str, Any]]:
            """Delegate to registry for OpenAI function calling format."""
            return self._registry.get_tool_definitions()

        def __len__(self) -> int:
            return len(self._registry)

        def __contains__(self, name: str) -> bool:
            return name in self._registry
    ```
