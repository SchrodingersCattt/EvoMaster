class MCPToolManager:

        self.tool_include_only: dict[str, list[str]] = {}
        self.loop: asyncio.AbstractEventLoop | None = None

    async def add_server(self, name: str, transport: str, **connection_kwargs) -> None:
        """连接 MCP server、获取 tools、构建 tool 信息。不含 runner/reconnection/progress。"""
        # 使用 create_connection 工厂
        # asyncio.wait_for(timeout=MCP_CONNECT_TIMEOUT) 包装 __aenter__ 和 list_tools
        # 失败时 3 次重试（保留 RESEARCH Pitfall 3 建议）
        # 成功后调用 _build_tools

    def _build_tools(self, name: str, conn: MCPConnection, tools_info: list) -> None:
        """从 list_tools 结果构建 tool 信息字典。"""
        # 从 evomaster MCPToolManager._build_tools 提取核心逻辑:
        # 1. tool_include_only filter（如果该 server 有白名单）
        # 2. sync_tools filter
        # 3. dedup（同名 tool 只保留第一个 server 的）

        #
        # 关键区别: 不创建 MCPTool 实例，存储轻量级 dict:
        # {tool_name: {"name": str, "description": str, "input_schema": dict,
        #              "remote_tool_name": str, "connection": conn,
        #              "has_path_adaptor": bool}}

    async def cleanup(self) -> None:
        """关闭所有 MCP 连接。"""
        # 遍历 _conn_ctxs 调用 __aexit__
        # 异常只 log.warning，不 raise
