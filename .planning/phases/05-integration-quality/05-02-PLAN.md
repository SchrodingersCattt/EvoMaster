         ```python
         if not isinstance(event, ToolResultEvent):
             return
         if self._ssh_attached:
             return  # remote workspace, skip

         if now - self._last_check_time < self._debounce_seconds:
             return  # debounce

             return  # no changes
         self._last_snapshot = snapshot
         self._upload()
         ```
