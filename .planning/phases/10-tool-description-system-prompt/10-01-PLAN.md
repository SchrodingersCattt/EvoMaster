<!-- ToolRegistry.get_tool_definitions() -- sends to LLM -->
```python
def get_tool_definitions(self) -> list[dict[str, Any]]:
    return [
        {"type": "function", "function": {"name": t.name, "description": t.description, "parameters": t.json_schema}}
        for t in self._tools.values()
    ]
```

<!-- ContextBuilder._build_tools() -- renders in system prompt -->
```python
@staticmethod
def _build_tools(tool_registry: ToolRegistry) -> str:
    tools = tool_registry.all_tools
    lines = [f"- {tool.name}: {tool.description}" for tool in tools]
    return "# Available Tools\n\n" + "\n".join(lines)
```
