**REPLACE lines 176-227 with:**

```python
# -- Phase 1: Serial guard + pre_hook gate --
approved_tools: list[ToolCallData] = []
for tc in response.tool_calls:
    # Guard evaluation (before hooks)
    guard_result = guard_pipeline.evaluate(tc, turn, spec.max_turns)
    if not guard_result.allowed:
        # Blocked: notify hooks, then append ToolMessage error
        await run_guard_blocked(spec.hooks, tc, guard_result)
        blocked_content = f"BLOCKED: {guard_result.reason}"
        if guard_result.guidance:
            blocked_content += f"\n{guard_result.guidance}"
        messages.append(
            ToolMessage(
                tool_call_id=tc.id,
                tool_name=tc.name,
                content=blocked_content,
            )
        )
        continue

    # pre_tool_call hook (intercepting, short-circuit)
    action = await run_pre_tool_call(spec.hooks, tc)
    if action == HookAction.SKIP:
        messages.append(
            ToolMessage(
                tool_call_id=tc.id,
                tool_name=tc.name,
                content="Tool call skipped by hook.",
            )
        )
        continue
    approved_tools.append(tc)

# -- Phase 2: Parallel execution of approved tools (D-04, D-05) --
if approved_tools:
    async def _execute_tool(tc: ToolCallData) -> ToolResult:
        try:
            return await spec.tool_registry.execute(tc.name, tc.arguments)
        except Exception as e:
            logger.exception("Tool execution failed: %s", tc.name)
            return ToolResult(
                status="error",
                content=(
                    f"Error executing tool '{tc.name}': "
                    f"{type(e).__name__}: {e}"
                ),
            )

    results = await asyncio.gather(
        *[_execute_tool(tc) for tc in approved_tools],
        return_exceptions=True,
    )

    # -- Phase 3: Serial post-processing (preserve original order) --
    for tc, raw_result in zip(approved_tools, results):
        if isinstance(raw_result, BaseException):
            tool_result = ToolResult(
                status="error",
                content=(
                    f"Error executing tool '{tc.name}': "
                    f"{type(raw_result).__name__}: {raw_result}"
                ),
            )
        else:
            tool_result = raw_result
        messages.append(
            ToolMessage(
                tool_call_id=tc.id,
                tool_name=tc.name,
                content=tool_result.content,
            )
        )
        # post_tool_call hook (observation, all hooks called) (D-06)
        await run_post_tool_call(spec.hooks, tc, tool_result)
```

Key implementation notes:
- `_execute_tool` is defined inside the `if approved_tools:` block as a closure. It catches exceptions and converts them to ToolResult(status="error") per D-05.
- `asyncio.gather(*coros, return_exceptions=True)` per D-05: all tools run to completion regardless of individual failures.
- The `return_exceptions=True` means gather returns exception objects (not raises them). The Phase 3 loop does `isinstance(raw_result, BaseException)` to detect these.
- Phase 3 iterates `zip(approved_tools, results)` which preserves the original tool_calls order since asyncio.gather returns results in input order.
- `post_tool_call` hook is called per tool in the serial Phase 3 loop per D-06.
