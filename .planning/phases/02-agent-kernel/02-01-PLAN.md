  <action>
    Create matmaster/engine/guard_pipeline.py with:

    1. Module-level constants: LOOP_WINDOW = 5, LOOP_THRESHOLD = 2

    2. LoopDetectionGuard class (satisfies Guard Protocol):
       def __init__(self, window: int = LOOP_WINDOW, threshold: int = LOOP_THRESHOLD) -> None:
           self._window = window
           self._threshold = threshold

       def evaluate(self, ctx: GuardContext) -> GuardResult:
           Compute fingerprint: f"{ctx.tool_name}|{json.dumps(ctx.tool_args, sort_keys=True, ensure_ascii=False)}"
           (try/except TypeError/ValueError for non-serializable args, fallback to str(ctx.tool_args))
           Count occurrences of same fingerprint in ctx.recent_calls[-self._window:].
           If count >= self._threshold: return GuardResult(allowed=False, reason=f"Loop detected: '{ctx.tool_name}' called {count+1} times with identical arguments in the last {self._window} calls.", guidance="Try a different approach or modify the arguments. Repeating the same call will not produce different results.")
           Else: return GuardResult(allowed=True)

    3. GuardPipeline class:
       def __init__(self, external_guards: list[Guard] | None = None) -> None:
           self._loop_guard = LoopDetectionGuard()
           self._guards: list[Guard] = [self._loop_guard]
           if external_guards:
               self._guards.extend(external_guards)
           self._recent_calls: deque[RecentCall] = deque(maxlen=LOOP_WINDOW)

       def evaluate(self, tool_call: ToolCallData, current_turn: int, max_turns: int) -> GuardResult:
           Build GuardContext from tool_call fields + current_turn + max_turns + list(self._recent_calls).
           Iterate self._guards: call guard.evaluate(ctx). If result.allowed is False, return result immediately.
           If all pass: record call via self._record_call(tool_call), return GuardResult(allowed=True).

       def _record_call(self, tool_call: ToolCallData) -> None:
           self._recent_calls.append(RecentCall(
               tool_name=tool_call.name,
               tool_args=tool_call.arguments,
               call_id=tool_call.id,
               timestamp=time.monotonic(),
           ))
