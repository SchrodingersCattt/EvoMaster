**2. 添加两个模块级桥接函数（在 class AgentKernel 之前）：**
```python
def _sync_iterate_async(async_iter):
    """Bridge async iterator to sync iterator.

    Temporary bridge for Phase 13-16 transition period.
    Will be removed in Phase 17 when Kernel becomes fully async.
    """
    loop = asyncio.new_event_loop()
    try:
        while True:
            try:
                yield loop.run_until_complete(async_iter.__anext__())
            except StopAsyncIteration:
                break
    finally:
        loop.close()


def _sync_call_async(coro):
    """Bridge async coroutine to sync call.

    Temporary bridge for Phase 13-16 transition period.
    Will be removed in Phase 17 when Kernel becomes fully async.
    """
    loop = asyncio.new_event_loop()
    try:
        return loop.run_until_complete(coro)
    finally:
        loop.close()
```

**3. 修改 AgentKernel.run() 入口管理 provider context manager：**

```python
def run(self, spec, task, history=None, stop_event=None) -> KernelRunResult:
    # Bridge async provider lifecycle (temporary Phase 13-16)
    _bridge_loop = asyncio.new_event_loop()
    try:
        _bridge_loop.run_until_complete(spec.llm_provider.__aenter__())
    except Exception:
        _bridge_loop.close()
        raise

    try:
        return self._run_loop(spec, task, history, stop_event, _bridge_loop)
    finally:
        try:
            _bridge_loop.run_until_complete(
                spec.llm_provider.__aexit__(None, None, None)
            )
        finally:
            _bridge_loop.close()
```
