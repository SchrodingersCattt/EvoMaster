    ```python
    class TestEffectLevelWithRealBuiltinMeta:
        """Verify CapabilityPolicy works with real BUILTIN_META effect_level values.

        Gap closure: BUILTIN_META previously used 'external_effect' which did
        not match the 'external_write' check in DefaultCapabilityPolicy.
        """

        def test_builtin_meta_web_tools_have_external_write(self) -> None:
            """BUILTIN_META web tools now use 'external_write' matching policy check."""
            from matmaster.tools.tool_catalog import BUILTIN_META
            for tool_name in ("mm_web_search", "web_fetch", "monitor_job"):
                _plane, effect_level, _fast = BUILTIN_META[tool_name]
                assert effect_level == "external_write", (
                    f"{tool_name} effect_level={effect_level!r}, expected 'external_write'"
                )

        def test_deny_web_tool_via_builtin_meta_values(self) -> None:
            """Construct instance using BUILTIN_META values, verify policy deny."""
            from matmaster.tools.tool_catalog import BUILTIN_META
            plane, effect_level, fast = BUILTIN_META["mm_web_search"]
            instance = _make_instance(effect_level=effect_level, plane=plane)
            topology = RuntimeTopology(
                session_kind="local",
                control_root="/ctrl",
                workspace_root="/ws",
                active_planes=frozenset({ToolPlane.SESSION_SHELL, ToolPlane.SESSION_FS}),
            )
            policy = DefaultCapabilityPolicy()
            decision = policy.evaluate(topology, instance, {})
            assert decision.decision == "deny"
            assert decision.guidance is not None
            assert "external" in decision.reason.lower()
    ```

    注意：`_make_instance` helper 需要接受可选的 `plane` 参数。检查现有 helper 签名，如果不支持 plane 参数则扩展它：在 _make_instance 函数中增加 `plane: ToolPlane = ToolPlane.CONTROL_PLANE` 参数，传入 ToolBinding 构造。
